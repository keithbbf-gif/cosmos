live_value.audio_owner from a real GET /api/v1/cvm/pull after POST
/api/v1/cvm/push against restarted :8770. An exit code is not evidence.

Repo-rootable: locate the repo from this file
(Path(__file__).resolve().parents[1]) and insert cosmos/ + builds/cvm-dt

def _http(svc, method, path, obj=None, token=None, timeout=10):
    req = urllib.request.Request(
        "http://127.0.0.1:%s%s" % (svc.port, path),
        data=(json.dumps(obj).encode("utf-8") if obj is not None else None),
        method=method)
    if token is not None:
        if token != "":
            req.add_header("Authorization", "Bearer " + token)
    else:
        req.add_header("Authorization", "Bearer " + svc.token)
    if obj is not None:
        req.add_header("Content-Type", "application/json")
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            return resp.status, json.loads(resp.read().decode("utf-8"))
    except urllib.error.HTTPError as e:
        return e.code, json.loads(e.read().decode("utf-8"))


def test_phone_push_never_second_audio_owner():
    """The one test this slice ships: phone push never owns audio; pull reflects turn."""
    tmp = _scratch()
    paths = CosmosPaths(tmp)
    kernel = _FakeKernel(paths)
    _seed_phone_ticket(paths, owner="phone")

    body = snapshot_delta("phone-turn-1", "", {
        "device": {"status": "ok", "audio_route": "none",
                   "surface": "capture+playback"},
        "future_kind": {"status": "ok"},
    }, client_id=PHONE_ID)
    body["audio_owner"] = "phone"

    rec = cvm_store_push(kernel, body)
    pulled = _cvm_pull_response(kernel, PHONE_ID)
    phone = json.loads(paths.state("cvm", "phone.json").read_text(encoding="utf-8"))
    ticket = json.loads(paths.state("cvm", "pull.json").read_text(encoding="utf-8"))

    owners = {
        rec.get("audio_owner"),
        pulled.get("audio_owner"),
        ticket.get("audio_owner"),
        phone.get("audio_owner"),
    }
    owners.discard(None)

    replay = cvm_store_push(kernel, body)
    pulled2 = _cvm_pull_response(kernel, PHONE_ID)

    wrote_audio = paths.state("cvm", "audio.json").is_file()
    wrote_handoff = paths.state("cvm", "handoff.json").is_file()

    return (
        rec.get("claimed") is False
        and rec.get("audio_owner") == "desktop"
        and rec.get("idempotent") is False
        and rec.get("cursor_out")
        and pulled.get("pull") is True
        and pulled.get("audio_owner") == "desktop"
        and pulled.get("cursor") == rec.get("cursor_out")
        and pulled.get("cursor") != "desk-before"
        and pulled.get("tree_id") == TREE_ID
        and pulled.get("client_id") == PHONE_ID
        and "device" in (phone.get("stored") or list((phone.get("kinds") or {}).keys()))
        and "future_kind" not in (phone.get("kinds") or {})
        and phone.get("request_id") == "phone-turn-1"
        and phone.get("client_id") == PHONE_ID
        and "audio_owner" not in json.dumps(phone.get("kinds") or {})
        and phone.get("audio_owner") != "phone"
        and ticket.get("audio_owner") == "desktop"
        and ticket.get("claimed") is False
        and owners == {"desktop"}
        and replay.get("idempotent") is True
        and replay.get("audio_owner") == "desktop"
        and pulled2.get("audio_owner") == "desktop"
        and pulled2.get("cursor") == rec.get("cursor_out")
        and wrote_audio is False
        and wrote_handoff is False
    )

def test_bearer_and_http_push_then_pull():
    """Existing bearer role; HTTP POST /cvm/push then GET /cvm/pull."""
    td = Path(tempfile.mkdtemp(prefix="cvm-push-http-"))
    install(td, tree_id=TREE_ID)
    k = Kernel(td, worker="cvm-push-test")
    k.paths.config("api_token.txt").write_text("http-token\n", encoding="utf-8")
    atomic_json(k.paths.state("cvm", "pull.json"), {
        "cvm": 1,
        "tree_id": TREE_ID,
        "pull": True,
        "kinds": ["device"],
        "cursor": "http-before",
        "audio_owner": "phone",
        "voice_client_timeout_s": 70.0,
        "issued_epoch": 1787800000.0,
        "writer": "test-seed",
    })
    svc = Service(k, host="127.0.0.1", port=0)
    svc.serve_background()
    try:
        body = snapshot_delta("http-turn-1", "", {
            "device": {"status": "ok", "audio_route": "none"},
        }, client_id=PHONE_ID)
        body["audio_owner"] = "phone"
        no_auth, no_body = _http(svc, "POST", PUSH_PATH, body, token="")
        bad_tok, _ = _http(svc, "POST", PUSH_PATH, body, token="wrong")
        code, pushed = _http(svc, "POST", PUSH_PATH, body)
        pcode, pulled = _http(svc, "GET",
                              "/api/v1/cvm/pull?client_id=" + PHONE_ID)
        replay_code, replay = _http(svc, "POST", PUSH_PATH, body)
        live = {
            "audio_owner": pulled.get("audio_owner"),
            "cursor": pulled.get("cursor"),
            "push_cursor_out": pushed.get("cursor_out"),
            "push_audio_owner": pushed.get("audio_owner"),
            "claimed": pushed.get("claimed"),
            "replay_idempotent": replay.get("idempotent"),
            "replay_audio_owner": replay.get("audio_owner"),
        }
        ok = (
            no_auth == 401 and no_body.get("error") == "UNAUTHORIZED"
            and bad_tok == 401
            and code == 200
            and pcode == 200
            and replay_code == 200
            and pushed.get("audio_owner") == "desktop"
            and pushed.get("claimed") is False
            and pulled.get("audio_owner") == "desktop"
            and pulled.get("cursor") == pushed.get("cursor_out")
            and pulled.get("cursor") != "http-before"
            and pulled.get("tree_id") == TREE_ID
            and replay.get("idempotent") is True
            and replay.get("audio_owner") == "desktop"
            and not k.paths.state("cvm", "audio.json").is_file()
        )
        return ok, live
    finally:
        svc.shutdown()

def main() -> int:
    check("phone push never becomes a second audio owner "
          "(push→pull quotes desktop, cursor is the pushed turn)",
          test_phone_push_never_second_audio_owner)
    check("missing pull.json is seeded desktop, never phone",
          test_missing_pull_json_is_seeded_desktop_not_phone)
    check("POST /cvm/snapshot still the P3 helper (no owner stamp)",
          test_snapshot_path_still_lands_without_claiming_owner)
    check("push reuses snapshot BAD_SNAPSHOT on a bad envelope",
          test_push_reuses_snapshot_idempotency_and_bad_envelope)
    check("source composes _cvm_store_snapshot + _cvm_pull_response "
          "(no kinds-filter copy, no phone owner, no bts_, no drive lit)",
          test_source_composes_not_copies)

    live = None
    http_err = ""

    def _http_probe():
        nonlocal live, http_err
        try:
            ok, live = test_bearer_and_http_push_then_pull()
            return ok
        except Exception as e:                                        # noqa: BLE001
            http_err = "%s: %s" % (type(e).__name__, e)
            raise

    check("HTTP POST /cvm/push then GET /cvm/pull: bearer 401, "
          "audio_owner=desktop, pull cursor is the pushed turn",
          _http_probe)
    if http_err and not any(p for l, p, _ in RESULTS if "HTTP POST" in l):
        RESULTS[-1] = (RESULTS[-1][0], False, http_err)

    ok = all(p for _, p, _ in RESULTS)
    for label, passed, err in RESULTS:
        print(("PASS" if passed else "FAIL") + "  " + label
              + (("  " + err) if err else ""))
    if live:
        print("live_value: " + json.dumps(live, sort_keys=True))
    print("result: " + ("ok" if ok else "FAIL")
          + "  %s/%s" % (sum(1 for _, p, _ in RESULTS if p), len(RESULTS)))
    return 0 if ok else 1


if __name__ == "__main__":
    raise SystemExit(main())
